\begin{pregunta}[25 puntos]
Un móvil parte del reposo con aceleración constante de $2\,\text{m/s}^2$. Calculá la distancia recorrida a los 5 segundos y la velocidad instantánea en ese momento.
\begin{resolucion}
Con $v_0 = 0$: $x = \tfrac{1}{2}at^2 = \tfrac{1}{2}(2)(5)^2 = 25\,\text{m}$. La velocidad instantánea es $v = at = 2 \cdot 5 = 10\,\text{m/s}$.
\end{resolucion}
\end{pregunta}

\begin{pregunta}[25 puntos]
Se lanza un proyectil con velocidad inicial de $40\,\text{m/s}$ formando un ángulo de $30^\circ$ con la horizontal. Determiná el alcance máximo y la altura máxima alcanzada.
\begin{resolucion}
$v_{0x} = 40\cos(30^\circ) \approx 34{,}64\,\text{m/s}$, $v_{0y} = 40\sin(30^\circ) = 20\,\text{m/s}$. Alcance: $R = \dfrac{v_0^2 \sin(2\theta)}{g} \approx 141{,}3\,\text{m}$. Altura máxima: $h = \dfrac{v_{0y}^2}{2g} \approx 20{,}4\,\text{m}$.
\end{resolucion}
\end{pregunta}

\begin{pregunta}[25 puntos]
Un bloque de $4\,\text{kg}$ sobre un plano inclinado de $20^\circ$ sin rozamiento.
\begin{resolucion}
Aceleración: $a = g\sin(20^\circ) \approx 3{,}35\,\text{m/s}^2$. Normal: $N = mg\cos(20^\circ) \approx 36{,}83\,\text{N}$.
\end{resolucion}
\end{pregunta}

\begin{pregunta}[25 puntos]
Colisión perfectamente inelástica entre $m_1=2\,\text{kg}$ a $6\,\text{m/s}$ y $m_2=3\,\text{kg}$ en reposo.
\begin{resolucion}
Conservación de momento: $v_f = \dfrac{m_1 v_1}{m_1+m_2} = \dfrac{12}{5} = 2{,}4\,\text{m/s}$. Energía perdida: $\Delta E_c = \tfrac{1}{2}m_1v_1^2 - \tfrac{1}{2}(m_1+m_2)v_f^2 = 36 - 14{,}4 = 21{,}6\,\text{J}$.
\end{resolucion}
\end{pregunta}
